# exactly the same bytes.
#
# To regenerate (inside the same pinned debian:trixie image build-deb.sh uses, so the resolved
# wheels match this image's Python/platform tags):
#   docker run --rm -v "$PWD:/src:ro" debian@sha256:<digest from build-deb.sh> bash -c '
#     set -e
#     apt-get update -qq && apt-get install -y -qq --no-install-recommends python3 python3-venv >/dev/null
#     python3 -m venv /tmp/genv && /tmp/genv/bin/pip install -q --upgrade pip
#     mkdir -p /tmp/wheels
#     # --only-binary=:all: refuses a sdist outright instead of silently mis-parsing one below (a
#     # sdist is named <name>-<version>.tar.gz, with no 4th "-"-separated field to split the
#     # version from the ".tar.gz" extension the way a wheel's pyver/abi/platform fields let the
#     # loop below do it for <name>-<version>-<pyver>-<abi>-<platform>.whl). Every direct and
#     # transitive dependency here is required to ship a cp313 (or py3/none-any) wheel; if one ever
#     # doesn't, this fails loudly here instead of writing a corrupt version into requirements.txt.
#     /tmp/genv/bin/pip download --only-binary=:all: -d /tmp/wheels "mcp>=1.28,<2" "pillow>=10" "pyyaml>=6,<7"
#     cd /tmp/wheels
#     for f in *.whl; do
#       rest=${f#*-}; name=${f%%-*}; version=${rest%%-*}
#       echo "$(echo "$name" | tr _ -)==$version --hash=sha256:$(sha256sum "$f" | cut -d" " -f1)"
#     done | sort
#   '
